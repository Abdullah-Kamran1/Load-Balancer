# Phase 7: HAProxy 3.4 reference implementation.
#
# One container, three HTTP frontends over the SAME four-backend farm the
# custom proxy uses (backend-1..4:8000 on lb-net):
#
#   :8091  frontend fe_roundrobin    -> backend be_roundrobin    (balance roundrobin)
#   :8092  frontend fe_leastconn     -> backend be_leastconn     (balance leastconn)
#   :8093  frontend fe_source        -> backend be_source        (balance source)
#   :8404  stats page + ";csv" export (diagnostics)
#
# Only the frontend under test receives load, so one load balancer handles
# a run at a time (plan fair-comparison rule).
#
# Settings mirror the custom proxy where HAProxy has an equivalent:
#   custom proxy                          HAProxy
#   GET /health, success = HTTP 200       option httpchk GET /health HTTP/1.1
#                                         http-check expect status 200
#   LB_HEALTH_INTERVAL=1.0                default-server inter 1s
#   LB_HEALTH_TIMEOUT=1.0                 timeout check 1s
#   LB_HEALTH_FAILURES=3  (-> DOWN)       fall 3
#   LB_HEALTH_RECOVERIES=2 (-> UP)        rise 2
#   connect_timeout 5.0                   timeout connect 5s
#   request_timeout 30.0 (read + idle)    timeout client/server/http-request 30s
#   one upstream connection per request   option http-server-close
#   no upstream retry                     retries 0
#
# Deliberate, recorded differences (docs/haproxy_setup.md):
#   - HAProxy answers 503 when no server can be reached; the custom proxy
#     answers 502.
#   - HAProxy `leastconn` counts server-side CONNECTIONS (plus queued), not
#     in-flight requests like the custom least_active_requests strategy.
#   - HAProxy `source` hashes the client IP with its map-based hash; the
#     custom ip_hash uses SHA-256. Both are sticky but not identical.
#   - HAProxy is multi-threaded (default nbthread = detected CPUs); the
#     custom proxy is a single asyncio thread. Natural configurations are
#     kept for both; the difference is reported with benchmark results.

global
    log stdout format raw local0 notice
    maxconn 4096
    # Default nbthread (one per detected CPU) is intentional: each
    # implementation runs in its natural configuration.

defaults
    mode http
    log global
    option httplog
    option dontlognull
    # Keep client keep-alive (wrk reuses connections) while closing the
    # server-side connection after each response, exactly like the custom
    # proxy's one-upstream-connection-per-request model.
    option http-server-close
    # Suppress per-request access logs (benchmark volume); state changes
    # and errors are still logged.
    option dontlog-normal
    retries 0
    timeout connect 5s
    timeout client 30s
    timeout server 30s
    timeout http-request 30s
    timeout http-keep-alive 30s
    timeout check 1s
    default-server check inter 1s fall 3 rise 2

frontend fe_roundrobin
    bind :8091
    default_backend be_roundrobin

frontend fe_leastconn
    bind :8092
    default_backend be_leastconn

frontend fe_source
    bind :8093
    default_backend be_source

backend be_roundrobin
    balance roundrobin
    option httpchk GET /health HTTP/1.1
    http-check expect status 200
    server backend-1 backend-1:8000
    server backend-2 backend-2:8000
    server backend-3 backend-3:8000
    server backend-4 backend-4:8000

backend be_leastconn
    balance leastconn
    option httpchk GET /health HTTP/1.1
    http-check expect status 200
    server backend-1 backend-1:8000
    server backend-2 backend-2:8000
    server backend-3 backend-3:8000
    server backend-4 backend-4:8000

backend be_source
    balance source
    option httpchk GET /health HTTP/1.1
    http-check expect status 200
    server backend-1 backend-1:8000
    server backend-2 backend-2:8000
    server backend-3 backend-3:8000
    server backend-4 backend-4:8000

# Diagnostics: HTML page at /stats, machine-readable CSV at /stats;csv
# (server rows expose status UP/DOWN, stot, hrsp_2xx..5xx).
listen stats
    bind :8404
    mode http
    stats enable
    stats uri /stats
